cmake_minimum_required(VERSION 3.16)
# Standalone portable codec checks; deliberately not wired into the renderer.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    project(OrbiterDDSCodec LANGUAGES CXX)
    enable_testing()
endif()
add_library(orbiter-dds-codec STATIC dds_decoder.cpp)
target_compile_features(orbiter-dds-codec PUBLIC cxx_std_17)
target_include_directories(orbiter-dds-codec PUBLIC ${CMAKE_CURRENT_SOURCE_DIR})
add_executable(orbiter-dds-codec-tests test_dds_decoder.cpp)
target_link_libraries(orbiter-dds-codec-tests PRIVATE orbiter-dds-codec)
add_test(NAME orbiter-dds-codec COMMAND orbiter-dds-codec-tests)
add_executable(orbiter-dds-decode-file dds_decode_file.cpp)
target_link_libraries(orbiter-dds-decode-file PRIVATE orbiter-dds-codec)
